// DİKKAT: 'use Route' satırını buraya EKLEMEYİN, yukarıda zaten var.
use Illuminate\Support\Facades\DB;
use App\Models\User;
use App\Models\Iaa;
use App\Models\MusteriSikayeti;

Route::get('/sistem/puanlari-esitle', function () {
    $users = User::all();
    $log = [];
    $updatedCount = 0;

    foreach ($users as $user) {
        // 1. KULLANICININ İLİŞKİLERİNİ ÇEKELİM
        // Üyesi olduğu takımların ID'leri
        $takimIdleri = $user->takimlar()->pluck('takim_id');
        
        // Bizzat görevli olduğu projelerin ID'leri
        $squadProjeIdleri = $user->gorevliOlduguProjeler()->pluck('iaas.id');

        // 2. PROJELERDEN KAZANILAN PUAN (Takım + Squad)
        // DashboardController'daki mantığın aynısı:
        $projePuani = Iaa::where('durum', 'Tamamlandı')
            ->where('puan', '>', 0)
            ->where(function($q) use ($takimIdleri, $squadProjeIdleri) {
                $q->whereIn('atanan_takim_id', $takimIdleri)
                  ->orWhereIn('id', $squadProjeIdleri);
            })
            ->sum('puan');

        // 3. ŞİKAYET KAYITLARINDAN KAZANILAN PUAN
        $sikayetPuani = MusteriSikayeti::where('olusturan_kurul_uyesi_id', $user->id)
            ->where('kazanilan_puan', '>', 0)
            ->sum('kazanilan_puan');

        // 4. CEZA PUANLARI (Disiplin)
        $cezaPuani = DB::table('disciplinary_cases')
            ->where('user_id', $user->id)
            ->where('durum', 'Karar Verildi')
            ->sum('hesaplanan_puan');

        // 5. NET PUAN HESAPLAMA
        $yeniNetPuan = ($projePuani + $sikayetPuani) - $cezaPuani;

        // Puan eksiye düşmemeli
        if ($yeniNetPuan < 0) {
            $yeniNetPuan = 0;
        }

        // 6. GÜNCELLEME
        if ($user->toplam_puan != $yeniNetPuan) {
            $eskiPuan = $user->toplam_puan;
            $user->toplam_puan = $yeniNetPuan;
            $user->save();
            
            $updatedCount++;
            $log[] = "<strong>{$user->name}</strong>: $eskiPuan -> <strong>$yeniNetPuan</strong> " .
                     "<small>(Proje: $projePuani + Şikayet: $sikayetPuani - Ceza: $cezaPuani)</small>";
        }
    }

    echo "<div style='font-family: sans-serif; padding: 20px;'>";
    echo "<h1 style='color: green;'>✅ Tam Eşitleme Başarıyla Tamamlandı!</h1>";
    echo "<h3>Toplam Güncellenen Kullanıcı Sayısı: $updatedCount</h3>";
    echo "<div style='background: #f4f4f4; padding: 15px; border-radius: 5px; border: 1px solid #ddd; line-height: 1.6;'>";
    
    if (count($log) > 0) {
        echo implode("<br>", $log);
    } else {
        echo "Tüm kullanıcıların puanları zaten doğru, herhangi bir değişiklik yapılmadı.";
    }
    
    echo "</div>";
    echo "</div>";
});
